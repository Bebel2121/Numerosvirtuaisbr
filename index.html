// 🌐 SITE PREMIUM - LOGIN + CADASTRO + DESIGN CROMÁTICO const express = require('express'); const fs = require('fs'); const crypto = require('crypto'); const nodemailer = require('nodemailer'); const bodyParser = require('body-parser'); const session = require('express-session'); // ⚠️ CONFIGURAÇÕES IMPORTANTES (ALTERE AQUI!) const SUA_CHAVE_SECRETA = 'meu_site_premium_2026'; // Pode mudar para o que quiser // 📧 DADOS DO SEU E-MAIL (PARA ENVIAR CONFIRMAÇÕES) const EMAIL_USUARIO = 'seuemailaqui@gmail.com'; // 🔴 SEU E-MAIL AQUI const EMAIL_SENHA = 'suasenha_de_app_aqui'; // 🔴 SENHA DE APP DO GMAIL (explico abaixo como fazer) const EMAIL_SERVICO = 'gmail'; // Não mudar se for Gmail // ⚙️ INICIALIZAÇÃO const app = express(); app.use(bodyParser.urlencoded({ extended: true })); app.use(bodyParser.json()); app.use(session({ secret: SUA_CHAVE_SECRETA, resave: false, saveUninitialized: true, cookie: { secure: false, maxAge: 86400000 } // Sessão dura 1 dia })); // 📁 BANCO DE DADOS (salva os usuários cadastrados) const bancoPath = './usuarios.json'; let banco = { usuarios: [] }; if (fs.existsSync(bancoPath)) { try { banco = JSON.parse(fs.readFileSync(bancoPath)) } catch(e) { console.log('Banco novo criado') } } function salvarBanco() { fs.writeFileSync(bancoPath, JSON.stringify(banco, null, 2)) } // 📧 SISTEMA DE ENVIO DE E-MAIL const transporter = nodemailer.createTransport({ service: EMAIL_SERVICO, auth: { user: EMAIL_USUARIO, pass: EMAIL_SENHA } }); // 🔐 TELA DE LOGIN app.get('/', (req, res) => { if (req.session.logado) return res.redirect('/painel'); res.send(` Meu Site Premium | Login

MEU SITE

Acesse sua conta exclusiva

ou continue com

Não tem conta? Criar conta agora

`); }); // 📝 TELA DE CADASTRO app.get('/cadastro', (req, res) => { res.send(` Cadastro | Meu Site Premium

📝 CRIAR CONTA

Preencha seus dados corretamente

Já tem conta? Voltar para login

`); }); // 💾 SALVAR CADASTRO E ENVIAR E-MAIL DE ATIVAÇÃO app.post('/salvar-cadastro', (req, res) => { const { nome, email, senha } = req.body; const codigoAtivacao = crypto.randomBytes(20).toString('hex'); // Código único // Verifica se e-mail já existe if (banco.usuarios.find(u => u.email === email)) { return res.send(''); } // Cria usuário novo const novoUsuario = { id: Date.now(), nome: nome, email: email, senha: crypto.createHash('sha256').update(senha).digest('hex'), // Senha criptografada ativo: false, codigo: codigoAtivacao }; banco.usuarios.push(novoUsuario); salvarBanco(); // 📧 ENVIAR E-MAIL COM LINK DE ATIVAÇÃO const link = `https://${req.headers.host}/ativar/${codigoAtivacao}`; const mailConfig = { from: EMAIL_USUARIO, to: email, subject: '✅ Ative sua Conta - Meu Site Premium', html: `

Olá ${nome}! 👋

Obrigado por se cadastrar! Para ativar sua conta e liberar todo o acesso, clique no botão abaixo:

ATIVAR MINHA CONTA

Se não foi você que se cadastrou, apenas ignore este e-mail com segurança.

` }; transporter.sendMail(mailConfig, (erro) => { if (erro) { return res.send('
❌ Erro ao enviar e-mail. Tente novamente mais tarde.
'); } res.send(`

📩 QUASE TUDO PRONTO!

Enviamos um link de confirmação para:

${email}

Acesse sua caixa de entrada (ou pasta de SPAM) e clique no link para ativar.

`); }); }); // ✅ ATIVAR CONTA PELO LINK app.get('/ativar/:codigo', (req, res) => { const usuario = banco.usuarios.find(u => u.codigo === req.params.codigo); if (!usuario) return res.send('

❌ Código inválido ou expirado!

'); usuario.ativo = true; usuario.codigo = null; salvarBanco(); res.send(`

✅ CONTA ATIVADA COM SUCESSO!

Agora você já pode fazer login e aproveitar tudo.

IR PARA O LOGIN
`); }); // 🔑 PROCESSAR LOGIN app.post('/entrar', (req, res) => { const { email, senha } = req.body; const senhaHash = crypto.createHash('sha256').update(senha).digest('hex'); const usuario = banco.usuarios.find(u => u.email === email && u.senha === senhaHash); if (!usuario) return res.send(''); if (!usuario.ativo) return res.send(''); req.session.logado = true; req.session.usuario = usuario; res.redirect('/painel'); }); // 🖥️ PAINEL PRINCIPAL (SÓ QUEM ESTÁ LOGADO VÊ) app.get('/painel', (req, res) => { if (!req.session.logado) return res.redirect('/'); res.send(` Painel | Meu Site Premium

MEU SITE

Bem-vindo, ${req.session.usuario.nome}

🚀 ÁREA EXCLUSIVA

Aqui você coloca todo o conteúdo do seu site, pois só quem estiver logado consegue ver!

Meu Perfil

Configurações

`); }); // 🚪 SAIR DA CONTA app.get('/sair', (req, res) => { req.session.destroy(); res.redirect('/'); }); // 🚀 INICIAR SERVIDOR const PORTA = process.env.PORT || 3000; app.listen(PORTA, () => console.log(`✅ SITE ONLINE NA PORTA ${PORTA}`));
